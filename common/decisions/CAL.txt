		allowed = {

		}
		on_map_mode = map_and_decisions_view
		state_target = yes
		ai_will_do = {
			base =10
		}
		target_array = ROOT.core_states
		days_remove = 35
		cancel_trigger = {
			FROM = {
				NOT = {
					#is_owned_by = ROOT
					is_controlled_by = ROOT
				}
			}
		}
		available = {
			command_power > 25
			NOT = { has_country_flag = country_flag_cal_recruiting_civil_war_militias }
		}
		target_trigger = {
			FROM = {
				is_core_of = ROOT
				is_controlled_by = ROOT
				#is_fully_controlled_by = ROOT
				state_population_k > 50
				NOT = { has_state_flag = state_flag_cal_recruited_civil_war_militia }
			}
		}
		custom_cost_trigger = {
			#command_power > 25
		}
		complete_effect = {
			add_command_power = -25
			set_country_flag = country_flag_cal_recruiting_civil_war_militias
		}
		ai_will_do = {
			base = 10
			modifier = {
				set_temp_variable = { cal_militia_state_number_temp = FROM.state_population_k }#state_population_k }
				divide_temp_variable = { cal_militia_state_number_temp = 101 }
				round_temp_variable = cal_militia_state_number_temp
				add = cal_militia_state_number_temp
			}
		}
		remove_effect = {
			set_temp_variable = { cal_militia_state_number_temp = FROM.state_population_k }#state_population_k }
			divide_temp_variable = { cal_militia_state_number_temp = 101 }
			round_temp_variable = cal_militia_state_number_temp
			clamp_temp_variable = {
				var = cal_militia_state_number_temp
				max = 11
				min = 1
			}
			FROM = { 
				for_loop_effect = {
					end = cal_militia_state_number_temp
					random_list = {
						7 = {
							create_unit = {
								division = "name = \"Civil War Militia\" division_template = \"Civil War Militia\" start_manpower_factor = 0.8"
								owner = ROOT
								count = 1
							}
						}
						3 = {
							create_unit = {
								division = "name = \"Cavalry Militia\" division_template = \"Cavalry Militia\" start_manpower_factor = 0.8"
								owner = ROOT
								count = 1
							}
						}
					}
				}
				set_state_flag = state_flag_cal_recruited_civil_war_militia
			}
			custom_effect_tooltip = cal_create_x_militias_civilwar_tt
			clr_country_flag = country_flag_cal_recruiting_civil_war_militias
		}
		cancel_if_not_visible = yes
		cancel_effect = {
			clr_country_flag = country_flag_cal_recruiting_civil_war_militias
		}
	}
