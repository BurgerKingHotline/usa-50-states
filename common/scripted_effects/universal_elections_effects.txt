# =============================================================================
# UNIVERSAL ELECTION SYSTEM - PRODUCTION BACKEND
# =============================================================================
# The demographic values are an equal-weight GOVERNMENT APPROVAL INDEX.
# They are not independent candidate vote shares. At election resolution the
# approval index produces a bounded incumbent swing, after which the highest
# party-popularity group wins. Country adapters may replace this resolution.
#
# Lifecycle flags:
#   universal_election_system_initialized = persistent backend initialized
#   is_in_election_season                 = final 180 days of the cycle
#   election_gui_visible                  = player explicitly opened election UI
#
# Momentum frames: 1 positive, 2 neutral, 3 negative.
# Bloc IDs: 0 Workers, 1 Farmers, 2 Military, 3 Wealthy,
#           4 White Collar, 5 Minorities, 6 Women.
# =============================================================================

migrate_universal_election_system = {
	if = {
		limit = { NOT = { has_country_flag = universal_election_system_initialized } }
		if = {
			limit = { NOT = { has_country_flag = wsh_election_arrays_initialized } }
			init_universal_election_system = yes
		}
		else = {
			set_country_flag = universal_election_system_initialized
			set_variable = { universal_election_schema_version = 3 }
			clr_country_flag = congress_visible
			clr_country_flag = election_gui_visible
			clr_country_flag = election_dropdown_open
			clr_country_flag = wsh_election_dropdown_open
			if = { limit = { NOT = { has_variable = days_until_election } } set_variable = { days_until_election = 1440 } }
			if = { limit = { NOT = { has_variable = election_actions_remaining } } set_variable = { election_actions_remaining = 0 } }
			initialize_universal_action_array = yes
			clear_array = elec_bloc_previous_support
			add_to_array = { elec_bloc_previous_support = elec_bloc_support^0 }
			add_to_array = { elec_bloc_previous_support = elec_bloc_support^1 }
			add_to_array = { elec_bloc_previous_support = elec_bloc_support^2 }
			add_to_array = { elec_bloc_previous_support = elec_bloc_support^3 }
			add_to_array = { elec_bloc_previous_support = elec_bloc_support^4 }
			add_to_array = { elec_bloc_previous_support = elec_bloc_support^5 }
			add_to_array = { elec_bloc_previous_support = elec_bloc_support^6 }
		}
	}
	else_if = {
		limit = {
			OR = {
				NOT = { has_variable = universal_election_schema_version }
				check_variable = { universal_election_schema_version < 3 }
			}
		}
		set_variable = { universal_election_schema_version = 3 }
		initialize_universal_action_array = yes
		clr_country_flag = election_dropdown_open
		clr_country_flag = wsh_election_dropdown_open
		if = { limit = { NOT = { has_variable = election_actions_remaining } } set_variable = { election_actions_remaining = 0 } }
		if = { limit = { NOT = { has_variable = elec_bloc_previous_support^6 } } clear_array = elec_bloc_previous_support add_to_array = { elec_bloc_previous_support = elec_bloc_support^0 } add_to_array = { elec_bloc_previous_support = elec_bloc_support^1 } add_to_array = { elec_bloc_previous_support = elec_bloc_support^2 } add_to_array = { elec_bloc_previous_support = elec_bloc_support^3 } add_to_array = { elec_bloc_previous_support = elec_bloc_support^4 } add_to_array = { elec_bloc_previous_support = elec_bloc_support^5 } add_to_array = { elec_bloc_previous_support = elec_bloc_support^6 } }
	}
}

initialize_election_cycle_effect = {
	if = { limit = { NOT = { has_country_flag = universal_election_system_initialized } } migrate_universal_election_system = yes }
	if = { limit = { NOT = { has_variable = days_until_election } } set_variable = { days_until_election = 1440 } }
}


synchronize_election_lifecycle_on_load = {
	initialize_election_cycle_effect = yes
	if = { limit = { check_variable = { days_until_election > 1440 } } set_variable = { days_until_election = 1440 } }
	if = {
		limit = { check_variable = { days_until_election < 1 } }
		resolve_universal_election_effect = yes
	}
	else_if = {
		limit = {
			check_variable = { days_until_election < 181 }
			NOT = { has_country_flag = is_in_election_season }
		}
		enter_election_season_effect = yes
	}
	else_if = {
		limit = {
			check_variable = { days_until_election > 180 }
			has_country_flag = is_in_election_season
		}
		clr_country_flag = is_in_election_season
		clr_country_flag = election_gui_visible
		clr_country_flag = election_dropdown_open
		set_variable = { election_actions_remaining = 0 }
	}
	set_variable = { election_countdown_frame = { value = days_until_election divide = 14.4 clamp = { min = 1 max = 100 } round = yes } }
}

enter_election_season_effect = {
	set_country_flag = is_in_election_season
	set_variable = { election_actions_remaining = 1 }
	clear_variable = selected_bloc_idx
	clr_country_flag = election_gui_visible
	clr_country_flag = election_dropdown_open
	clr_country_flag = wsh_election_dropdown_open
}

apply_incumbent_approval_swing_effect = {
	# Approval above/below 50 creates a maximum +/-10 percentage-point swing.
	set_temp_variable = { election_incumbent_swing = gov_national_vote_share }
	subtract_from_temp_variable = { election_incumbent_swing = 50 }
	multiply_temp_variable = { election_incumbent_swing = 0.002 }
	set_temp_variable = { election_incumbent_swing = { value = election_incumbent_swing clamp = { min = -0.10 max = 0.10 } } }
	if = { limit = { has_government = democratic } add_popularity = { ideology = democratic popularity = election_incumbent_swing } }
	else_if = { limit = { has_government = conservatism_ideology } add_popularity = { ideology = conservatism_ideology popularity = election_incumbent_swing } }
	else_if = { limit = { has_government = socialism_ideology } add_popularity = { ideology = socialism_ideology popularity = election_incumbent_swing } }
	else_if = { limit = { has_government = communism } add_popularity = { ideology = communism popularity = election_incumbent_swing } }
	else_if = { limit = { has_government = fascism } add_popularity = { ideology = fascism popularity = election_incumbent_swing } }
	else_if = { limit = { has_government = populism_ideology } add_popularity = { ideology = populism_ideology popularity = election_incumbent_swing } }
	else_if = { limit = { has_government = neutrality } add_popularity = { ideology = neutrality popularity = election_incumbent_swing } }
}

resolve_generic_party_election_effect = {
	apply_incumbent_approval_swing_effect = yes
	set_temp_variable = { election_winner_id = 0 }
	set_temp_variable = { election_winner_score = party_popularity@democratic }
	if = { limit = { check_variable = { party_popularity@conservatism_ideology > election_winner_score } } set_temp_variable = { election_winner_id = 1 } set_temp_variable = { election_winner_score = party_popularity@conservatism_ideology } }
	if = { limit = { check_variable = { party_popularity@socialism_ideology > election_winner_score } } set_temp_variable = { election_winner_id = 2 } set_temp_variable = { election_winner_score = party_popularity@socialism_ideology } }
	if = { limit = { check_variable = { party_popularity@communism > election_winner_score } } set_temp_variable = { election_winner_id = 3 } set_temp_variable = { election_winner_score = party_popularity@communism } }
	if = { limit = { check_variable = { party_popularity@fascism > election_winner_score } } set_temp_variable = { election_winner_id = 4 } set_temp_variable = { election_winner_score = party_popularity@fascism } }
	if = { limit = { check_variable = { party_popularity@populism_ideology > election_winner_score } } set_temp_variable = { election_winner_id = 5 } set_temp_variable = { election_winner_score = party_popularity@populism_ideology } }
	if = { limit = { check_variable = { party_popularity@neutrality > election_winner_score } } set_temp_variable = { election_winner_id = 6 } set_temp_variable = { election_winner_score = party_popularity@neutrality } }
	set_variable = { last_election_winner_id = election_winner_id }
	if = { limit = { check_variable = { election_winner_id = 0 } } set_politics = { ruling_party = democratic elections_allowed = no } }
	else_if = { limit = { check_variable = { election_winner_id = 1 } } set_politics = { ruling_party = conservatism_ideology elections_allowed = no } }
	else_if = { limit = { check_variable = { election_winner_id = 2 } } set_politics = { ruling_party = socialism_ideology elections_allowed = no } }
	else_if = { limit = { check_variable = { election_winner_id = 3 } } set_politics = { ruling_party = communism elections_allowed = no } }
	else_if = { limit = { check_variable = { election_winner_id = 4 } } set_politics = { ruling_party = fascism elections_allowed = no } }
	else_if = { limit = { check_variable = { election_winner_id = 5 } } set_politics = { ruling_party = populism_ideology elections_allowed = no } }
	else = { set_politics = { ruling_party = neutrality elections_allowed = no } }
}

resolve_universal_election_effect = {
	# Elections terminate pre-election coalitions; the winner forms the next government.
	if = { limit = { has_country_flag = uses_wsh_legislature has_coalition = yes } end_coalition = yes }

	# Country-specific election adapters own special candidate/leadership logic.
	if = {
		limit = { has_country_flag = uses_chg_council }
		CHG_resolve_leadership_vote = yes
	}
	else = {
		resolve_generic_party_election_effect = yes
	}

	# Synchronize each institution only after the winning government is known.
	if = {
		limit = { has_country_flag = uses_wsh_legislature }
		create_coalitions_array_if_missing = yes
		default_party_popularity_PP_gain = yes
		WSH_recalculate_all_seats = yes
	}
	if = {
		limit = { has_country_flag = uses_chg_council }
		CHG_calculate_and_refresh_council = yes
	}

	add_to_variable = { elections_held = 1 }
	if = {
		limit = { is_ai = no has_country_flag = uses_chg_council }
		country_event = { id = ffl_election.2 }
	}
	else_if = {
		limit = { is_ai = no }
		country_event = { id = ffl_election.1 }
	}
	set_variable = { days_until_election = 1440 }
	set_variable = { election_countdown_frame = 100 }
	set_variable = { election_actions_remaining = 0 }
	clr_country_flag = is_in_election_season
	clr_country_flag = election_gui_visible
	clr_country_flag = election_dropdown_open
	clr_country_flag = wsh_election_dropdown_open
	clear_variable = selected_bloc_idx
}

universal_ai_campaign_effect = {
	if = {
		limit = {
			is_ai = yes
			check_variable = { election_actions_remaining > 0 }
			NOT = { check_variable = { political_power < 15 } }
		}
		add_political_power = -15
		random_list = {
			1 = { set_variable = { elec_bloc_support^0 = { value = elec_bloc_support^0 add = 2 clamp = { min = 1 max = 100 } } } }
			1 = { set_variable = { elec_bloc_support^1 = { value = elec_bloc_support^1 add = 2 clamp = { min = 1 max = 100 } } } }
			1 = { set_variable = { elec_bloc_support^2 = { value = elec_bloc_support^2 add = 2 clamp = { min = 1 max = 100 } } } }
			1 = { set_variable = { elec_bloc_support^3 = { value = elec_bloc_support^3 add = 2 clamp = { min = 1 max = 100 } } } }
			1 = { set_variable = { elec_bloc_support^4 = { value = elec_bloc_support^4 add = 2 clamp = { min = 1 max = 100 } } } }
			1 = { set_variable = { elec_bloc_support^5 = { value = elec_bloc_support^5 add = 2 clamp = { min = 1 max = 100 } } } }
			1 = { set_variable = { elec_bloc_support^6 = { value = elec_bloc_support^6 add = 2 clamp = { min = 1 max = 100 } } } }
		}
		set_variable = { election_actions_remaining = 0 }
		refresh_election_display_state = yes
	}
}

universal_monthly_election_tick_effect = {
	initialize_election_cycle_effect = yes
	subtract_from_variable = { days_until_election = 30 }

	if = {
		limit = {
			check_variable = { days_until_election < 181 }
			NOT = { has_country_flag = is_in_election_season }
		}
		enter_election_season_effect = yes
	}

	if = {
		limit = { has_country_flag = is_in_election_season }
		set_variable = { election_actions_remaining = 1 }
		universal_election_dynamic_drift_effect = yes
		if = { limit = { has_country_flag = uses_wsh_legislature } wsh_calculate_demographic_subideology_weights = yes }
		universal_ai_campaign_effect = yes
	}

	if = {
		limit = { check_variable = { days_until_election < 1 } }
		resolve_universal_election_effect = yes
	}
	else = {
		set_variable = { election_countdown_frame = { value = days_until_election divide = 14.4 clamp = { min = 1 max = 100 } round = yes } }
		refresh_election_display_state = yes
	}
}
