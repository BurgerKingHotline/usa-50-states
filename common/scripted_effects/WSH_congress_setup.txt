WSH_apportion_house_seats = {
	set_variable = { wsh_conservative_house_seats = 0 }
	set_variable = { wsh_agrarian_house_seats = 0 }
	set_variable = { wsh_liberal_house_seats = 0 }
	set_variable = { wsh_socialist_house_seats = 0 }
	set_variable = { wsh_communist_house_seats = 0 }
	set_variable = { wsh_fascist_house_seats = 0 }
	set_variable = { wsh_nonaligned_house_seats = 0 }
	set_temp_variable = { wsh_total_popularity = party_popularity@conservatism_ideology }
	add_to_temp_variable = { wsh_total_popularity = party_popularity@populism_ideology }
	add_to_temp_variable = { wsh_total_popularity = party_popularity@democratic }
	add_to_temp_variable = { wsh_total_popularity = party_popularity@socialism_ideology }
	add_to_temp_variable = { wsh_total_popularity = party_popularity@communism }
	add_to_temp_variable = { wsh_total_popularity = party_popularity@fascism }
	add_to_temp_variable = { wsh_total_popularity = party_popularity@neutrality }
	set_temp_variable = { wsh_apportion_i = 0 }
	if = {
		limit = { check_variable = { wsh_total_popularity < 0.001 } }
		set_variable = { wsh_nonaligned_house_seats = 98 }
		set_temp_variable = { wsh_apportion_i = 98 }
	}
	while_loop_effect = {
		limit = { check_variable = { wsh_apportion_i < 98 } }
		set_temp_variable = { wsh_divisor = { value = wsh_conservative_house_seats add = 1 } }
		set_temp_variable = { wsh_best_score = { value = party_popularity@conservatism_ideology divide = wsh_divisor } }
		set_temp_variable = { wsh_winner = 0 }
		set_temp_variable = { wsh_divisor = { value = wsh_agrarian_house_seats add = 1 } }
		set_temp_variable = { wsh_test_score = { value = party_popularity@populism_ideology divide = wsh_divisor } }
		if = { limit = { check_variable = { wsh_test_score > wsh_best_score } } set_temp_variable = { wsh_winner = 1 } set_temp_variable = { wsh_best_score = wsh_test_score } }
		set_temp_variable = { wsh_divisor = { value = wsh_liberal_house_seats add = 1 } }
		set_temp_variable = { wsh_test_score = { value = party_popularity@democratic divide = wsh_divisor } }
		if = { limit = { check_variable = { wsh_test_score > wsh_best_score } } set_temp_variable = { wsh_winner = 2 } set_temp_variable = { wsh_best_score = wsh_test_score } }
		set_temp_variable = { wsh_divisor = { value = wsh_socialist_house_seats add = 1 } }
		set_temp_variable = { wsh_test_score = { value = party_popularity@socialism_ideology divide = wsh_divisor } }
		if = { limit = { check_variable = { wsh_test_score > wsh_best_score } } set_temp_variable = { wsh_winner = 3 } set_temp_variable = { wsh_best_score = wsh_test_score } }
		set_temp_variable = { wsh_divisor = { value = wsh_communist_house_seats add = 1 } }
		set_temp_variable = { wsh_test_score = { value = party_popularity@communism divide = wsh_divisor } }
		if = { limit = { check_variable = { wsh_test_score > wsh_best_score } } set_temp_variable = { wsh_winner = 4 } set_temp_variable = { wsh_best_score = wsh_test_score } }
		set_temp_variable = { wsh_divisor = { value = wsh_fascist_house_seats add = 1 } }
		set_temp_variable = { wsh_test_score = { value = party_popularity@fascism divide = wsh_divisor } }
		if = { limit = { check_variable = { wsh_test_score > wsh_best_score } } set_temp_variable = { wsh_winner = 5 } set_temp_variable = { wsh_best_score = wsh_test_score } }
		set_temp_variable = { wsh_divisor = { value = wsh_nonaligned_house_seats add = 1 } }
		set_temp_variable = { wsh_test_score = { value = party_popularity@neutrality divide = wsh_divisor } }
		if = { limit = { check_variable = { wsh_test_score > wsh_best_score } } set_temp_variable = { wsh_winner = 6 } set_temp_variable = { wsh_best_score = wsh_test_score } }
		if = { limit = { check_variable = { wsh_winner = 0 } } add_to_variable = { wsh_conservative_house_seats = 1 } }
		else_if = { limit = { check_variable = { wsh_winner = 1 } } add_to_variable = { wsh_agrarian_house_seats = 1 } }
		else_if = { limit = { check_variable = { wsh_winner = 2 } } add_to_variable = { wsh_liberal_house_seats = 1 } }
		else_if = { limit = { check_variable = { wsh_winner = 3 } } add_to_variable = { wsh_socialist_house_seats = 1 } }
		else_if = { limit = { check_variable = { wsh_winner = 4 } } add_to_variable = { wsh_communist_house_seats = 1 } }
		else_if = { limit = { check_variable = { wsh_winner = 5 } } add_to_variable = { wsh_fascist_house_seats = 1 } }
		else = { add_to_variable = { wsh_nonaligned_house_seats = 1 } }
		add_to_temp_variable = { wsh_apportion_i = 1 }
	}
}

	if = { limit = { check_variable = { coalitions_array^3 > 1 } } add_to_temp_variable = { coalition_senate_count = wsh_communist_senate_seats } add_to_temp_variable = { coalition_house_count = wsh_communist_house_seats } }
	if = { limit = { check_variable = { coalitions_array^4 > 1 } } add_to_temp_variable = { coalition_senate_count = wsh_fascist_senate_seats } add_to_temp_variable = { coalition_house_count = wsh_fascist_house_seats } }
	if = { limit = { check_variable = { coalitions_array^5 > 1 } } add_to_temp_variable = { coalition_senate_count = wsh_agrarian_senate_seats } add_to_temp_variable = { coalition_house_count = wsh_agrarian_house_seats } }
	if = { limit = { check_variable = { coalitions_array^6 > 1 } } add_to_temp_variable = { coalition_senate_count = wsh_nonaligned_senate_seats } add_to_temp_variable = { coalition_house_count = wsh_nonaligned_house_seats } }
	set_variable = { coalition_senate_seats = coalition_senate_count }
	set_variable = { coalition_house_seats = coalition_house_count }
	set_variable = { coalition_senate_majority_frame = { value = coalition_senate_count divide = 0.38 clamp = { min = 1 max = 100 } round = yes } }
	set_variable = { coalition_house_majority_frame = { value = coalition_house_count divide = 0.98 clamp = { min = 1 max = 100 } round = yes } }
	set_variable = { coalition_controls_senate = 0 }
	set_variable = { coalition_controls_house = 0 }
	if = { limit = { check_variable = { coalition_senate_count > 19 } } set_variable = { coalition_controls_senate = 1 } }
	if = { limit = { check_variable = { coalition_house_count > 49 } } set_variable = { coalition_controls_house = 1 } }
}

	WSH_house_reload = yes

WSH_update_legislative_governance = {
	WSH_refresh_legislature_display = yes
	set_variable = { wsh_legislative_pp_gain = -0.10 }
	if = {
		limit = {
			OR = {
				check_variable = { coalition_controls_senate = 1 }
				check_variable = { coalition_controls_house = 1 }
			}
		}
		set_variable = { wsh_legislative_pp_gain = 0 }
	}
	if = {
		limit = {
			check_variable = { coalition_controls_senate = 1 }
			check_variable = { coalition_controls_house = 1 }
		}
		set_variable = { wsh_legislative_pp_gain = 0.10 }
	}
	if = {
		limit = { NOT = { has_dynamic_modifier = { modifier = WSH_legislative_control_modifier } } }
		add_dynamic_modifier = { modifier = WSH_legislative_control_modifier }
	}
	force_update_dynamic_modifier = yes
}

	WSH_calculate_party_seats = yes
	WSH_update_legislative_governance = yes
