    desc = california_major_events.12.desc
	fire_only_once = yes
	trigger = {
		#tag = CAL
		has_country_flag = country_flag_cal_socialist_civil_war
		NOT = { has_idea = CAL_idea_guerrilla_fighting }
	}
    option = {
        name = california_major_events.12.a
		add_ideas = { CAL_idea_guerrilla_fighting }

			if = {
				limit = {
					NOT = {
						is_owned_by = ROOT
					}
				}
				remove_core_of = ROOT
				add_claim_by = ROOT
			}
		}
    }
}
