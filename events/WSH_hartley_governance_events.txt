# 50. Convergence event - fires on completion of WSH_first_republican_budget
country_event = {
	id = WSH_hartley.50
	title = WSH_hartley.50.t
	desc = WSH_hartley.50.desc
	picture = GFX_report_event_usa_election_generic

	is_triggered_only = yes
	fire_only_once = yes

	immediate = {
		if = {
			limit = {
				has_country_flag = WSH_guard_joint_board
				has_country_flag = WSH_property_individual_transfer
				has_country_flag = WSH_relief_county_grants
			}
			add_ideas = WSH_legislative_compromise_government
			set_country_flag = WSH_hartley_government_legislative_compromise
		}
		else_if = {
			limit = { has_country_flag = WSH_treasury_cut_expenditure }
			add_ideas = WSH_constitutional_conservative_government
			set_country_flag = WSH_hartley_government_constitutional_conservative
		}
		else_if = {
			limit = { has_country_flag = WSH_treasury_emergency_bonds }
			add_ideas = WSH_emergency_development_government
			set_country_flag = WSH_hartley_government_emergency_development
		}
		else = {
			add_ideas = WSH_corporate_reconstruction_government
			set_country_flag = WSH_hartley_government_corporate_reconstruction
		}
		set_country_flag = WSH_next_election_preparations
	}
	option = {
		name = WSH_hartley.50.a
		add_stability = 0.05
	}
}
