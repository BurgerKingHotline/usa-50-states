# ---- First treaties (WSH_treaty_and_neighbor_relations)

country_event = {
	id = WSH_diplomacy.1
	title = WSH_diplomacy.1.t
	desc = WSH_diplomacy.1.desc
	picture = GFX_report_event_usa_election_generic
	is_triggered_only = yes
	fire_only_once = yes

	option = { #Columbia River agreement with Oregon
		name = WSH_diplomacy.1.a
		set_country_flag = WSH_treaty_oregon
		add_to_variable = { WSH_public_works_commitment = 1 }
		if = {
			limit = { has_country_flag = WSH_martin_economic_line }
			add_to_variable = { WSH_public_works_commitment = 1 }
			add_to_variable = { WSH_state_capacity = 1 }
		}
	}
	option = { #Northern trade agreement with British Columbia
		name = WSH_diplomacy.1.b
		set_country_flag = WSH_treaty_canada
		add_to_variable = { WSH_business_confidence = 1 }
		if = {
			limit = { has_country_flag = WSH_private_credit_system }
			add_to_variable = { WSH_business_confidence = 1 }
		}
		if = {
			limit = { has_country_flag = WSH_state_nationalization }
			add_to_variable = { WSH_business_confidence = -1 }
		}
	}
	option = { #Grain and rail agreement with Idaho
		name = WSH_diplomacy.1.c
		set_country_flag = WSH_treaty_idaho
		add_to_variable = { WSH_county_autonomy = -1 }
		add_to_variable = { WSH_unemployment_pressure = -1 }
	}
	option = { #Pacific shipping agreement with California
		name = WSH_diplomacy.1.d
		set_country_flag = WSH_treaty_california
		add_to_variable = { WSH_port_autonomy = 1 }
		add_to_variable = { WSH_business_confidence = 1 }
	}
	option = { #Federal-restoration talks
		name = WSH_diplomacy.1.e
		set_country_flag = WSH_treaty_federal_talks
		add_to_variable = { WSH_democratic_legitimacy = 1 }
		add_to_variable = { WSH_military_autonomy = -1 }
	}
}

# ---- The Klan and the New Order of Cincinnatus
# Fires 180 days after the auxiliaries are accepted; the Klan takes the New Order
# only if its influence has kept growing.
country_event = {
	id = WSH_klan.1
	title = WSH_klan.1.t
	desc = WSH_klan.1.desc
	picture = GFX_report_event_usa_election_generic
	is_triggered_only = yes
	fire_only_once = yes
	trigger = {
		has_country_flag = WSH_klan_auxiliaries
		check_variable = { WSH_klan_influence > 4 }
	}

	option = {
		name = WSH_klan.1.a
		recruit_character = WASHINGTON_LUTHER_POWELL
		set_party_name = {
			ideology = fascism
			long_name = WSH_fascism_party_klan_long
			name = WSH_fascism_party_klan
		}
		set_country_flag = WSH_klan_controls_new_order
		add_to_variable = { WSH_klan_influence = 1 }
		add_to_variable = { WSH_democratic_legitimacy = -1 }
	}
}
